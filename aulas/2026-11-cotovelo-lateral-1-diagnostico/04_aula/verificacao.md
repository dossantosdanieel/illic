- Usar até 79% como teto do pré-teste — s21 (dado e notas).

- Com PRTEE inicial alto, preferir a régua proporcional — s34 (notas).

import sys

from pptx.enum.chart import XL_TICK_LABEL_POSITION
p = Presentation(sys.argv[1])
for sl in p.slides:
    for sh in sl.shapes:
        if getattr(sh, "has_chart", False) and sh.has_chart:
            vals = [v for pl in sh.chart.plots for se in pl.series for v in se.values if v is not None]

                sh.chart.category_axis.tick_label_position = XL_TICK_LABEL_POSITION.LOW
p.save(sys.argv[1])
